"""
Utility functions for factorial calculation and data processing.
"""

from typing import Iterable, List


def calculate_factorial(n: int) -> int:
    """Return the factorial of a non‑negative integer *n*.

    Raises:
        ValueError: If *n* is negative.
    """
    if n < 0:
        raise ValueError("Factorial is not defined for negative numbers")
    result = 1
    for i in range(1, n + 1):
        result *= i
    return result


def process_data(data: List[int]) -> List[int]:
    """Return a new list with all zero values removed from *data*.

    The original list is not modified.
    """
    return [item for item in data if item != 0]


def sum_positive_numbers(numbers: Iterable[int]) -> int:
    """Return the sum of *numbers* if all are positive, otherwise 0."""
    if all(n > 0 for n in numbers):
        return sum(numbers)
    return 0


if __name__ == "__main__":
    print(calculate_factorial(5))
